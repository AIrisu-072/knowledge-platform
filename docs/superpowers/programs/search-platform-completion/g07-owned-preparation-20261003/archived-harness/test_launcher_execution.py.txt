def approval(phase="bootstrap"):
    value = deepcopy(APPROVAL)
    if phase == "supervise":
        value.update(phase=phase, run_root=ROOT, test_executable={
            "path": TEST_EXE, "sha256": HASHES["test_executable_sha256"],
        })
    elif phase == "bootstrap-resume":
        value.update(phase=phase, run_root=ROOT)
    return value


def request(stage="runtime-red"):
    value = {"schema_version": 1, "outcome": "GO", "run_id": RUN_ID,
             "stage": stage, **HASHES, "cargo_window_statement_id": None}

class ApprovalDecisionTests(unittest.TestCase):
    def check(self, document, phase="bootstrap", approved=APPROVAL_DIGEST,
              observed=APPROVAL_DIGEST):
        return launcher.validate_approval(document, phase, approved, observed)

    def test_bootstrap_bound_to_independent_approval_digest_is_accepted(self):
        self.assertIs(self.check(approval()), True)

    def test_supervise_with_exact_owned_root_and_executable_is_accepted(self):
        self.assertIs(self.check(approval("supervise"), "supervise"), True)

    def test_bootstrap_resume_with_exact_owned_root_and_null_test_is_accepted(self):
        self.assertIs(self.check(approval("bootstrap-resume"), "bootstrap-resume"), True)

    def test_bootstrap_resume_requires_canonical_owned_root(self):
        for root in (None, ROOT + "/../other", "/tmp/p6pg.ABCDEFGH"):
            value = {**approval("bootstrap-resume"), "run_root": root}
            with self.subTest(root=root):
                self.assertIs(self.check(value, "bootstrap-resume"), False)

    def test_bootstrap_resume_rejects_nonnull_test_executable(self):
        value = {**approval("bootstrap-resume"), "test_executable": {
            "path": TEST_EXE, "sha256": HASHES["test_executable_sha256"],
        }}
        self.assertIs(self.check(value, "bootstrap-resume"), False)

    def test_self_asserted_approval_digest_field_is_rejected(self):
        self.assertIs(self.check({**approval(), "approval_sha256": APPROVAL_DIGEST}), False)

    def test_arbitrary_command_and_url_fields_are_rejected(self):
        for key in ("argv", "url"):
            with self.subTest(key=key):
                self.assertIs(self.check({**approval(), key: "synthetic-unapproved"}), False)

    def test_missing_approval_field_is_rejected(self):
        for key in APPROVAL:
            value = approval()
            del value[key]
            with self.subTest(key=key):
                self.assertIs(self.check(value), False)

    def test_boolean_schema_version_is_rejected(self):
        self.assertIs(self.check({**approval(), "schema_version": True}), False)

    def test_wrong_approval_outcome_or_scope_is_rejected(self):
        for key, value in (("outcome", "NO-GO"), ("scope", "another scope")):
            with self.subTest(key=key):
                self.assertIs(self.check({**approval(), key: value}), False)

    def test_independent_phase_mismatch_is_rejected(self):
        self.assertIs(self.check(approval(), "supervise"), False)

    def test_unrecognized_phase_is_rejected(self):
        self.assertIs(self.check({**approval(), "phase": "quality"}, "quality"), False)

    def test_independent_approval_digest_mismatch_is_rejected(self):
        self.assertIs(self.check(approval(), observed="8" * 64), False)

    def test_malformed_independent_approval_digest_is_rejected(self):

            with self.subTest(key=key):

        receipt, expected, actual, registry = receipt_fixture()

            altered = deepcopy(expected)
            del altered[key]
            with self.subTest(context_missing=key):
                self.assertIs(launcher.validate_test_receipt(receipt, altered, actual, registry), False)

    def test_nested_database_and_child_keys_must_be_exact(self):

            for field in ("database", "children"):
                nested = original[field] if field == "database" else original[field][0]
                for key in nested:
                    altered = deepcopy(original)
                    target = altered[field] if field == "database" else altered[field][0]
                    del target[key]
                    args = (altered, expected, actual, registry) if container_name == "receipt" else (receipt, altered, actual, registry)
                    with self.subTest(container=container_name, field=field, missing=key):
                        self.assertIs(launcher.validate_test_receipt(*args), False)
                altered = deepcopy(original)
                target = altered[field] if field == "database" else altered[field][0]
                target["unexpected"] = True
                args = (altered, expected, actual, registry) if container_name == "receipt" else (receipt, altered, actual, registry)
                with self.subTest(container=container_name, field=field, extra=True):
                    self.assertIs(launcher.validate_test_receipt(*args), False)

class ElfInspectionDecisionTests(unittest.TestCase):
    def test_exact_executable_elf_closure_is_reduced_without_execution(self):
        self.assertEqual(launcher.inspect_elf_text(ELF_TEXT, deepcopy(ELF_CLOSURE)), ELF_CLOSURE)

    def test_shared_library_without_interpreter_is_accepted(self):
        closure = {**deepcopy(ELF_CLOSURE), "interpreter": None}
        value = ELF_TEXT.replace(ELF_INTERPRETER, b"").replace(b"DYN (Position-Independent Executable file)", b"DYN (Shared object file)")
        self.assertEqual(launcher.inspect_elf_text(value, closure), closure)

    def test_same_version_name_from_two_exact_providers_is_accepted(self):
        value = ELF_TEXT.replace(b"contains 2 entries:", b"contains 3 entries:")

                              b" 0x0000000000000001 (NEEDED)             Shared library: [libm.so.6]\n 0x0000000000000000 (NULL)               0x0\n")
        value = value.replace(b"section '.gnu.version_r' contains 1 entry:",
                              b"section '.gnu.version_r' contains 2 entries:")

        value += b"  0x0040:   Name: GLIBC_2.2.5  Flags: none  Version: 4\n"
        closure = {
            "interpreter": ELF_CLOSURE["interpreter"], "needed": ["libc.so.6", "libm.so.6"],
            "version_needs": ["libc.so.6:GLIBC_2.2.5", "libc.so.6:GLIBC_2.34", "libm.so.6:GLIBC_2.2.5"],
        }
        self.assertEqual(launcher.inspect_elf_text(value, closure), closure)

    def test_wrong_elf_class_endianness_and_machine_are_rejected(self):
        for old, new in ((b"ELF64", b"ELF32"), (b"little endian", b"big endian"), (b"Advanced Micro Devices X86-64", b"AArch64")):
            with self.subTest(old=old):
                self.assertEqual(launcher.inspect_elf_text(ELF_TEXT.replace(old, new), deepcopy(ELF_CLOSURE)), "REJECT")

    def test_unknown_or_duplicate_interpreter_is_rejected(self):
        for value in (ELF_TEXT.replace(b"/lib64/ld-linux-x86-64.so.2", b"/tmp/alternate-loader"), ELF_TEXT + ELF_INTERPRETER):
            with self.subTest(value=value):
                self.assertEqual(launcher.inspect_elf_text(value, deepcopy(ELF_CLOSURE)), "REJECT")

    def test_extra_missing_and_duplicate_needed_entries_are_rejected(self):
        needed = b" 0x0000000000000001 (NEEDED)             Shared library: [libc.so.6]\n"
        for value in (ELF_TEXT + needed.replace(b"libc.so.6", b"libunknown.so.1"), ELF_TEXT.replace(needed, b""), ELF_TEXT + needed):
            with self.subTest(value=value):
                self.assertEqual(launcher.inspect_elf_text(value, deepcopy(ELF_CLOSURE)), "REJECT")

    def test_unapproved_or_missing_version_need_is_rejected(self):
        for value in (ELF_TEXT.replace(b"GLIBC_2.34", b"GLIBC_99.0"), ELF_TEXT.replace(b"  0x0020:   Name: GLIBC_2.34  Flags: none  Version: 3\n", b"")):
            with self.subTest(value=value):
                self.assertEqual(launcher.inspect_elf_text(value, deepcopy(ELF_CLOSURE)), "REJECT")

    def test_version_need_cannot_be_attributed_to_wrong_library(self):
        value = ELF_TEXT.replace(b"File: libc.so.6", b"File: libunknown.so.1")
        self.assertEqual(launcher.inspect_elf_text(value, deepcopy(ELF_CLOSURE)), "REJECT")

    def test_rpath_runpath_and_textrel_are_rejected(self):
        for tag in (b"RPATH", b"RUNPATH", b"TEXTREL"):
            with self.subTest(tag=tag):
                value = ELF_TEXT + b" 0x000000000000001d (" + tag + b") Library runpath: [/tmp]\n"
                self.assertEqual(launcher.inspect_elf_text(value, deepcopy(ELF_CLOSURE)), "REJECT")

    def test_closure_unknown_or_missing_key_is_rejected(self):
        self.assertEqual(launcher.inspect_elf_text(ELF_TEXT, {**deepcopy(ELF_CLOSURE), "rpath": []}), "REJECT")
        self.assertEqual(launcher.inspect_elf_text(ELF_TEXT, {key: value for key, value in ELF_CLOSURE.items() if key != "version_needs"}), "REJECT")

    def test_closure_duplicate_or_empty_requirement_is_rejected(self):
        for key, value in (("needed", ["libc.so.6", "libc.so.6"]), ("needed", [""]), ("version_needs", ["libc.so.6:GLIBC_2.2.5", "libc.so.6:GLIBC_2.2.5"])):
            with self.subTest(key=key, value=value):
                self.assertEqual(launcher.inspect_elf_text(ELF_TEXT, {**deepcopy(ELF_CLOSURE), key: value}), "REJECT")

    def test_malformed_or_empty_elf_text_is_rejected(self):
        for value in (b"", b"synthetic ordinary text\n", ELF_TEXT.replace(b"Shared library: [libc.so.6]", b"Shared library: [libc.so.6")):
            with self.subTest(value=value):
                self.assertEqual(launcher.inspect_elf_text(value, deepcopy(ELF_CLOSURE)), "REJECT")

    def test_nonbytes_or_non_ascii_elf_text_is_rejected(self):
        self.assertEqual(launcher.inspect_elf_text(ELF_TEXT.decode(), deepcopy(ELF_CLOSURE)), "REJECT")
        self.assertEqual(launcher.inspect_elf_text(ELF_TEXT + b"\xff", deepcopy(ELF_CLOSURE)), "REJECT")

    def test_oversized_elf_text_is_rejected(self):
        self.assertEqual(launcher.inspect_elf_text(b"x" * 65537, deepcopy(ELF_CLOSURE)), "REJECT")
        # Only ELF inspection gains the 256-KiB cap; retain the malformed-input
        # rejection above and exercise valid text across the old/new boundaries.
        for size in (65537, 131072, 234448, 262144):
            padded = ELF_TEXT + b"\n" + b" " * (size - len(ELF_TEXT) - 2) + b"\n"
            with self.subTest(valid_elf_bytes=size):
                self.assertEqual(launcher.inspect_elf_text(padded, deepcopy(ELF_CLOSURE)), ELF_CLOSURE)
        oversized = ELF_TEXT + b"\n" + b" " * (262145 - len(ELF_TEXT) - 2) + b"\n"
        self.assertEqual(launcher.inspect_elf_text(oversized, deepcopy(ELF_CLOSURE)), "REJECT")


def synthetic_listing(*paths):
    listing = mock.MagicMock()
    listing.__enter__.return_value = [SimpleNamespace(path=path) for path in paths]
    listing.__exit__.return_value = False
    return listing


class DiskTreeRetryTests(unittest.TestCase):
    def test_one_missing_entry_retries_complete_tree_and_discards_partial_count(self):
        root = Path(ROOT)
        partial, vanished, stable = root / "partial", root / "vanished", root / "stable"
        directory = SimpleNamespace(st_mode=stat.S_IFDIR, st_size=1024, st_blocks=0)
        large_file = SimpleNamespace(st_mode=stat.S_IFREG, st_size=65536, st_blocks=0)
        stable_file = SimpleNamespace(st_mode=stat.S_IFREG, st_size=4096, st_blocks=0)

        def observed_lstat(path):
            if path == root:
                return directory
            if path == partial:
                return large_file
            if path == vanished:
                raise FileNotFoundError("synthetic entry vanished")
            if path == stable:
                return stable_file
            self.fail("unexpected synthetic traversal: " + str(path))

        watch = launcher.DiskWatch(SimpleNamespace(path=root), None)
        # Traversal is LIFO: the first scan counts partial before missing-entry
        # failure. The second complete scan has only root+stable = 5120 bytes.
        with mock.patch.object(launcher.Path, "lstat", autospec=True, side_effect=observed_lstat) as lstat_mock, \
                mock.patch.object(launcher.os, "scandir", side_effect=[
                    synthetic_listing(str(vanished), str(partial)), synthetic_listing(str(stable)),
                ]) as scan_mock, \
                mock.patch.object(launcher.time, "monotonic", return_value=9.0):
            try:
                measured = watch.tree_bytes(root, 10.0)
            except launcher.PreparationStop as error:
                self.fail("first missing-entry race needs one complete retry: " + str(error))
            self.assertEqual(measured, 5120)
            self.assertEqual(scan_mock.call_count, 2)
            self.assertEqual(sum(call.args[0] == root for call in lstat_mock.call_args_list), 2)

    def test_second_missing_entry_stops_after_one_complete_tree_retry(self):
        root, vanished = Path(ROOT), Path(ROOT) / "vanished"
        directory = SimpleNamespace(st_mode=stat.S_IFDIR, st_size=1024, st_blocks=0)

        def observed_lstat(path):
            if path == root:
                return directory
            raise FileNotFoundError("synthetic entry vanished again")

        watch = launcher.DiskWatch(SimpleNamespace(path=root), None)
        with mock.patch.object(launcher.Path, "lstat", autospec=True, side_effect=observed_lstat) as lstat_mock, \
                mock.patch.object(launcher.os, "scandir", side_effect=[
                    synthetic_listing(str(vanished)), synthetic_listing(str(vanished)),
                ]) as scan_mock, \
                mock.patch.object(launcher.time, "monotonic", return_value=9.0):
            with self.assertRaises(launcher.PreparationStop):
                watch.tree_bytes(root, 10.0)
            self.assertEqual(scan_mock.call_count, 2)
            self.assertEqual(sum(call.args[0] == root for call in lstat_mock.call_args_list), 2)

    def test_retry_cannot_restart_original_deadline_before_second_scan(self):
        root, vanished, stable = Path(ROOT), Path(ROOT) / "vanished", Path(ROOT) / "stable"
        directory = SimpleNamespace(st_mode=stat.S_IFDIR, st_size=1024, st_blocks=0)
        regular = SimpleNamespace(st_mode=stat.S_IFREG, st_size=4096, st_blocks=0)
        elapsed = [False]
        observed_times = []

        def observed_lstat(path):
            if path == root:
                return directory
            if path == vanished:
                elapsed[0] = True
                raise FileNotFoundError("synthetic race at original deadline")
            return regular

        def observed_time():
            value = 10.001 if elapsed[0] else 9.0
            observed_times.append(value)
            return value

        watch = launcher.DiskWatch(SimpleNamespace(path=root), None)
        with mock.patch.object(launcher.Path, "lstat", autospec=True, side_effect=observed_lstat) as lstat_mock, \
                mock.patch.object(launcher.os, "scandir", side_effect=[
                    synthetic_listing(str(vanished)), synthetic_listing(str(stable)),
                ]) as scan_mock, \
                mock.patch.object(launcher.time, "monotonic", side_effect=observed_time):
            with self.assertRaises(launcher.PreparationStop):
                watch.tree_bytes(root, 10.0)
            self.assertIn(10.001, observed_times)
            self.assertEqual(scan_mock.call_count, 1)
            self.assertEqual(sum(call.args[0] == root for call in lstat_mock.call_args_list), 1)

    def test_free_space_floor_is_checked_before_incomplete_growth_snapshot(self):
        root = SimpleNamespace(path=Path(ROOT), verify=mock.Mock())
        policy = SimpleNamespace(disk_budget_ok=lambda free, run, target, baseline: free >= launcher.FLOOR)
        watch = launcher.DiskWatch(root, policy)
        free = SimpleNamespace(f_bavail=launcher.FLOOR - 1, f_frsize=1)
        with mock.patch.object(launcher.time, "monotonic", return_value=9.0), \
                mock.patch.object(launcher.os, "statvfs", return_value=free), \
                mock.patch.object(watch, "tree_bytes", side_effect=launcher.PreparationStop("synthetic incomplete growth snapshot")) as tree_mock, \
                mock.patch.object(launcher.Path, "iterdir", side_effect=AssertionError("unexpected extra traversal")):
            with self.assertRaises(launcher.PreparationStop):
                watch.check(force=True)
            tree_mock.assert_not_called()
