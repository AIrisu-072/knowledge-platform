def digest(path):
    with path.open('rb') as f:
        return hashlib.file_digest(f, 'sha256').hexdigest()


def save(path, data):
    path.write_text(json.dumps(data, indent=2) + '\n')


def tree_size(path):
    if not path.exists():
        return 0
    total = 0
    for root, dirs, files in os.walk(path, followlinks=False):
        if any(Path(root, n).is_symlink() for n in dirs + files):
            raise RuntimeError('symlink in owned measured tree')
        for name in files:
            try:
                total += Path(root, name).stat().st_size
            except FileNotFoundError:
                continue  # atomic Cargo replacement; next sample counts it
    return total


def owned_growth():
    # Conservative accounting covers all retained G07 planning/source/receipt
    # files under the existing evidence root, including earlier failed runs.
    evidence = BASE / 'search-recovery-evidence'
    total = tree_size(DRAFT) + tree_size(TARGET)

            if path.is_symlink():
                raise RuntimeError('unexpected G07 evidence symlink')
            total += tree_size(path) if path.is_dir() else path.stat().st_size
    return total


def proc_stat(pid):
    text = Path('/proc', str(pid), 'stat').read_text()
    end = text.rfind(')')
    fields = text[end + 2:].split()
    if end < 0 or len(fields) < 20:
        raise RuntimeError('invalid process stat')
    return {'pid': pid, 'state': fields[0], 'pgrp': int(fields[2]),
            'session': int(fields[3]), 'start_ticks': int(fields[19])}


def session_members(identity):
    # Read only stat identity metadata, never argv/environ/credentials.
    # This observes the original session only. The trusted compiler closure
    # must not daemonize; this is not hostile/detached-process containment.
    members = []
    for item in Path('/proc').iterdir():
        if not item.name.isdigit():
            continue
        try:
            s = proc_stat(int(item.name))
        except FileNotFoundError:
            continue
        except PermissionError:
            raise RuntimeError('process membership unreadable')
        if s['session'] == identity['pid']:
            if s['start_ticks'] < identity['start_ticks'] or s['pgrp'] != identity['pid']:
                raise RuntimeError('unexpected owned-session membership')
            if s['state'] not in ('Z', 'X'):
                members.append(s)
    return members


def peek_exit(p):
    return os.waitid(os.P_PID, p.pid, os.WEXITED | os.WNOHANG | os.WNOWAIT)


def same_leader(identity):
    now = proc_stat(identity['pid'])
    if any(now[k] != identity[k] for k in ('pid', 'start_ticks', 'pgrp', 'session')):
        raise RuntimeError('owned leader identity changed')


def stop_group(p, identity, deadline):
    # WNOWAIT retains the leader (even a zombie) until the owned group is done,
    # preventing PID/PGID reuse before any signal. Descendants are only observed
    # non-running; this parent never claims it reaped grandchildren.
    for sig, seconds in ((signal.SIGTERM, 3), (signal.SIGKILL, 3)):
        same_leader(identity)
        if not session_members(identity):
            break
        os.killpg(p.pid, sig)
        until = min(deadline, time.monotonic() + seconds)
        while time.monotonic() < until:
            if not session_members(identity):
                break
            time.sleep(.1)
    if session_members(identity):
        UNCERTAIN.append(p)
        return 'owned-group-exit-unknown'
    if peek_exit(p) is None:
        UNCERTAIN.append(p)
        return 'leader-exit-unknown'
    p.wait(timeout=max(.001, min(1, deadline - time.monotonic())))
    return None


def run_step(label, argv, cwd, env, cap, directory):
    started = time.monotonic()
    deadline = started + cap
    work_deadline = deadline - min(12, cap * .8)
    output_cap = 4 * 1024**2 if label == 'rust-compile' else 64 * 1024
    result = {'label': label, 'argv': list(map(str, argv)), 'cwd': str(cwd), 'deadline_seconds': cap,
              'returncode': None, 'abnormal_reason': None, 'logs': {}, 'environment_keys': sorted(env), 'output_cap_per_stream': output_cap}
    save(directory / (label + '.initial.json'), result)
    paths = {k: directory / (label + '.' + k + '.log') for k in ('stdout', 'stderr')}
    p, identity, threads, errors = None, None, [], []
    overflow = threading.Event()
    minimum_free = shutil.disk_usage(BASE).free
    def drain(pipe, path):
        retained = 0
        try:
            with path.open('xb') as f:
                while chunk := pipe.read(65536):
                    take = min(len(chunk), output_cap - retained)
                    if take:
                        f.write(chunk[:take]); retained += take
                    if take < len(chunk):
                        overflow.set()
        except Exception as e:
            errors.append(type(e).__name__ + ': ' + str(e))
        finally:
            pipe.close()
    try:
        if minimum_free < FLOOR or owned_growth() > GROWTH:
            raise RuntimeError('disk budget before command')
        p = subprocess.Popen(argv, cwd=cwd, env=env, stdin=subprocess.DEVNULL,
                             stdout=subprocess.PIPE, stderr=subprocess.PIPE, start_new_session=True)
        identity = proc_stat(p.pid)
        if identity['pgrp'] != p.pid or identity['session'] != p.pid:
            raise RuntimeError('new owned session missing')
        result['leader_identity'] = identity
        save(directory / (label + '.spawn.json'), result)
        for name, pipe in (('stdout', p.stdout), ('stderr', p.stderr)):
            t = threading.Thread(target=drain, args=(pipe, paths[name]), daemon=True)
            t.start(); threads.append(t)
        next_disk = started
        while True:
            now = time.monotonic()
            same_leader(identity)
            exited = peek_exit(p) is not None
            if exited and not session_members(identity):
                break
            if now >= next_disk:
                free = shutil.disk_usage(BASE).free; minimum_free = min(minimum_free, free)
                if free < FLOOR or owned_growth() > GROWTH:
                    raise RuntimeError('disk budget')
                next_disk = now + 1
            if now >= work_deadline:
                raise RuntimeError('timeout')
            if overflow.is_set() or errors:
                raise RuntimeError('output-overflow-or-drainer-error')
            time.sleep(.1)
    except BaseException as e:
        result['abnormal_reason'] = type(e).__name__ + ': ' + str(e)
    finally:
        if p is not None:
            try:
                if identity is None:
                    # The handle still owns this unreaped child; retrieve only
                    # that identity before any group signal. Failure is Unknown.
                    identity = proc_stat(p.pid)
                same_leader(identity)
                if session_members(identity):
                    cleanup_error = stop_group(p, identity, deadline - 3)
                    if cleanup_error:
                        raise RuntimeError(cleanup_error)
                elif peek_exit(p) is not None:
                    p.wait(timeout=max(.001, min(1, deadline - time.monotonic())))
                else:
                    raise RuntimeError('leader-exit-unknown')
            except BaseException as e:
                UNCERTAIN.append(p)
                result['abnormal_reason'] = (result['abnormal_reason'] or '') + '; cleanup: ' + type(e).__name__ + ': ' + str(e)
            drain_deadline = min(deadline, time.monotonic() + 3)
            for t in threads:
                t.join(max(0, drain_deadline - time.monotonic()))
            if any(t.is_alive() for t in threads) or errors or overflow.is_set():
                result['abnormal_reason'] = (result['abnormal_reason'] or '') + '; output incomplete/overflow: ' + repr(errors)
            result['returncode'] = p.returncode
        result['elapsed_seconds'] = round(time.monotonic() - started, 3)
        result['minimum_free_bytes'] = minimum_free
        try:
            for name, path in paths.items():
                if path.exists():
                    result['logs'][name] = {'path': str(path), 'sha256': digest(path), 'bytes': path.stat().st_size}
            save(directory / (label + '.terminal.json'), result)
        except BaseException as e:
            print('TERMINAL_RECEIPT_WRITE_UNKNOWN', label, type(e).__name__, flush=True)
            raise
    if result['abnormal_reason']:
        raise RuntimeError('command failed: ' + result['abnormal_reason'])
    return result


def combined(result):
    return '\n'.join(Path(v['path']).read_text(errors='replace') for v in result['logs'].values())


def verify_inputs(admission, directory, aggregate_deadline=None):

        raise RuntimeError('wrong admission scope')
    expected = {str(p) for p in INPUTS + TOOLS}
    if set(admission.get('sha256', {})) != expected:
        raise RuntimeError('admission must pin exactly fixed inputs/tools')
    for p in INPUTS + TOOLS:
        if digest(p) != admission['sha256'][str(p)]:
            raise RuntimeError('reviewed input mismatch: ' + str(p))
    # Fixed read-only Git commands validate the entire tracked closure against
    # the pinned base, allowing only the two separately hash-pinned test edits.
    env = {'HOME': str(directory / 'home'), 'LC_ALL': 'C', 'TZ': 'UTC', 'PATH': '/usr/bin:/bin'}
    git = [str(TOOLS[3]), '-c', 'core.fsmonitor=false', '-c', 'core.untrackedCache=false']
    queries = [('head', ['rev-parse', 'HEAD']), ('tree', ['rev-parse', 'HEAD^{tree}']),
               ('dirty', ['diff', '--no-ext-diff', '--no-textconv', '--name-only', 'HEAD', '--']),
               ('untracked', ['ls-files', '--others', '--exclude-standard'])]
    values = {}
    tag = str(time.monotonic_ns())
    for name, args in queries:
        cap = 15 if aggregate_deadline is None else min(15, aggregate_deadline - time.monotonic())
        if cap <= 0:
            raise RuntimeError('pure aggregate deadline before source-closure command')
        r = run_step('git-' + tag + '-' + name, git + args, REPO, env, cap, directory)
        if r['returncode'] != 0:
            raise RuntimeError('read-only source closure check failed')
        values[name] = Path(r['logs']['stdout']['path']).read_text().strip()
    if values['head'] != HEAD or values['tree'] != TREE:
        raise RuntimeError('base HEAD/tree changed')
    if set(values['dirty'].splitlines()) - ALLOWED_DIRTY or values['untracked']:
        raise RuntimeError('source closure exceeds two-file scope')
