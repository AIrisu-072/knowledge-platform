clock sleeps or launcher entrypoint. The sole local import is the reviewed stub.

failures from None-returning stubs. Do not interpret that as runtime admission.
"""

class ContainmentTests(unittest.TestCase):
    def test_owned_regular_path_is_accepted(self):
        self.assertIs(policy.owned_path_is_safe(ROOT, ROOT + "/data/PG_VERSION", OBS, UID), True)

    def test_prefix_lookalike_is_rejected(self):
        self.assertIs(policy.owned_path_is_safe(ROOT, ROOT + "-other/data", OBS, UID), False)

    def test_parent_traversal_is_rejected(self):
        self.assertIs(policy.owned_path_is_safe(ROOT, ROOT + "/data/../outside", OBS, UID), False)

    def test_symlink_component_is_rejected_even_if_target_is_inside(self):
        observed = {**OBS, ROOT + "/data": {"kind": "symlink", "uid": UID, "mode": 0o777}}
        self.assertIs(policy.owned_path_is_safe(ROOT, ROOT + "/data/PG_VERSION", observed, UID), False)

    def test_unknown_path_observation_is_rejected(self):
        self.assertIs(policy.owned_path_is_safe(ROOT, ROOT + "/missing", OBS, UID), False)

    def test_non_owner_private_root_is_rejected(self):
        observed = {**OBS, ROOT: {"kind": "directory", "uid": 999, "mode": 0o700}}
        self.assertIs(policy.owned_path_is_safe(ROOT, ROOT + "/data", observed, UID), False)

    def test_group_readable_run_root_is_rejected(self):
        observed = {**OBS, ROOT: {"kind": "directory", "uid": UID, "mode": 0o750}}
        self.assertIs(policy.owned_path_is_safe(ROOT, ROOT + "/data", observed, UID), False)

class DeadlineAndDiskTests(unittest.TestCase):
    def test_phase_uses_minimum_of_local_and_remaining_total(self):
        self.assertEqual(policy.phase_remaining(0.0, 89.0, 90.0, 88.0, 15.0), 1.0)

    def test_new_phase_cannot_restart_expired_total(self):
        self.assertEqual(policy.phase_remaining(0.0, 90.0, 90.0, 90.0, 15.0), 0.0)

    def test_backward_clock_observation_is_rejected(self):
        self.assertEqual(policy.phase_remaining(10.0, 9.0, 90.0, 10.0, 15.0), "REJECT")

    def test_nonfinite_clock_is_rejected(self):
        self.assertEqual(policy.phase_remaining(0.0, float("nan"), 90.0, 0.0, 15.0), "REJECT")

    def test_disk_floor_and_exact_growth_boundary_are_accepted(self):
        self.assertIs(policy.disk_budget_ok(1610612736, 2**30, 7 * 2**30, 0), True)

    def test_disk_floor_violation_is_rejected(self):
        self.assertIs(policy.disk_budget_ok(1610612735, 0, 0, 0), False)

    def test_cargo_target_growth_counts_against_eight_gib(self):
        self.assertIs(policy.disk_budget_ok(10 * 2**30, 2**30, 7 * 2**30 + 1, 0), False)

    def test_missing_or_decreasing_target_measurement_is_unknown(self):
        self.assertIs(policy.disk_budget_ok(10 * 2**30, 0, 99, 100), False)


class CommandSelectionTests(unittest.TestCase):
    def test_red_selects_only_last_claim_named_test(self):
        self.assertEqual(policy.test_command("runtime-red", TEST_EXE),
                         (TEST_EXE, "last_claim_kill9_restart_reaps_unknown_once", "--exact", "--nocapture", "--test-threads=1"))

    def test_first_green_selects_only_last_claim_named_test(self):
        self.assertEqual(policy.test_command("runtime-green-last", TEST_EXE),
                         (TEST_EXE, "last_claim_kill9_restart_reaps_unknown_once", "--exact", "--nocapture", "--test-threads=1"))

    def test_second_green_selects_only_four_process_named_test(self):
        self.assertEqual(policy.test_command("runtime-green-four", TEST_EXE),
                         (TEST_EXE, "four_processes_disjoint_claims_and_recover_expired", "--exact", "--nocapture", "--test-threads=1"))

    def test_unknown_stage_cannot_select_workspace_or_child_test(self):
        self.assertEqual(policy.test_command("child_worker_fixture", TEST_EXE), "REJECT")

    def test_relative_test_executable_is_rejected(self):
        self.assertEqual(policy.test_command("runtime-red", "cargo"), "REJECT")

    def test_stage_approval_cannot_repeat_red(self):
        self.assertIs(policy.admit_stage(HELPER_GREEN + ["runtime-red-confirmed"], "runtime-red", REVIEWS, True), False)

    def test_runtime_red_requires_helper_green_and_exact_reviewed_hashes(self):
        self.assertIs(policy.admit_stage(HELPER_GREEN, "runtime-red", REVIEWS, True), True)

    def test_helper_red_alone_cannot_admit_runtime_red(self):
        self.assertIs(policy.admit_stage(["compile-helper-red", "pure-rust-red-confirmed"], "runtime-red", REVIEWS, True), False)

    def test_missing_source_review_cannot_admit_runtime_red(self):
        reviews = {k: v for k, v in REVIEWS.items() if not k.startswith("source_")}
        self.assertIs(policy.admit_stage(HELPER_GREEN, "runtime-red", reviews, True), False)

    def test_missing_launcher_review_cannot_admit_runtime_red(self):
        reviews = {k: v for k, v in REVIEWS.items() if not k.startswith("launcher_")}
        self.assertIs(policy.admit_stage(HELPER_GREEN, "runtime-red", reviews, True), False)

    def test_changed_fixture_hash_cannot_admit_runtime_red(self):
        self.assertIs(policy.admit_stage(HELPER_GREEN, "runtime-red", {**REVIEWS, "fixture_observed": "e" * 64}, True), False)

    def test_green_requires_expected_red_and_fresh_compile(self):
        self.assertIs(policy.admit_stage(HELPER_GREEN + ["runtime-red-confirmed", "compile-runtime-green", "pure-rust-final-green-confirmed"], "runtime-green-last", REVIEWS, True), True)

    def test_green_is_rejected_after_terminal_failure(self):
        self.assertIs(policy.admit_stage(["compile-helper-red", "terminal-unknown"], "runtime-green-last", REVIEWS, True), False)

    def test_source_hash_mismatch_rejects_even_an_expected_stage(self):
        self.assertIs(policy.admit_stage(HELPER_GREEN, "runtime-red", {**REVIEWS, "source_observed": "e" * 64}, True), False)

    def test_dead_cluster_cannot_be_restarted_by_request(self):
        self.assertIs(policy.admit_stage(HELPER_GREEN, "runtime-red", REVIEWS, False), False)

            "P6_G07_CHILD_HANDSHAKE": HANDSHAKE, "P6_G07_TEST_LAUNCH_ID": LAUNCH_ID,

    def test_only_supplied_loopback_barrier_is_added(self):
        result = policy.child_environment(ROOT + "/scope.json", ROOT + "/home", ROOT + "/tmp", "reap", "11111111-1111-4111-8111-111111111111", 1, DATABASE, HANDSHAKE, LAUNCH_ID, "127.0.0.1:12345")

        self.assertEqual(result, expected)

    def test_archive_absolute_name_is_rejected(self):
        self.assertIs(policy.archive_member_is_safe({"name": "/usr/bin/m4", "kind": "file", "mode": 0o755, "link": None}, {}), False)

    def test_archive_parent_traversal_is_rejected(self):
        self.assertIs(policy.archive_member_is_safe({"name": "usr/../../escape", "kind": "file", "mode": 0o755, "link": None}, {}), False)

    def test_archive_setuid_is_rejected(self):
        self.assertIs(policy.archive_member_is_safe({"name": "usr/bin/m4", "kind": "file", "mode": 0o4755, "link": None}, {}), False)

    def test_archive_link_redirect_before_later_file_is_rejected(self):
        prior = {"usr": {"kind": "symlink", "link": "../../outside"}}
        self.assertIs(policy.archive_member_is_safe({"name": "usr/bin/m4", "kind": "file", "mode": 0o755, "link": None}, prior), False)

    def test_archive_cross_package_file_collision_is_rejected(self):
        prior = {"usr/bin/m4": {"kind": "file", "link": None}}
        self.assertIs(policy.archive_member_is_safe({"name": "usr/bin/m4", "kind": "file", "mode": 0o755, "link": None}, prior), False)

    def test_archive_shared_ordinary_directory_is_accepted(self):

        self.assertIs(policy.archive_member_is_safe({"name": "usr", "kind": "directory", "mode": 0o755, "link": None}, prior), True)


class OutcomeAndOutputTests(unittest.TestCase):
    def test_only_confirmed_unwired_normal_nonzero_red_is_expected(self):
        self.assertIs(policy.expected_red_is_confirmed(RED), True)

    def test_outer_timeout_cannot_be_expected_red(self):
        self.assertIs(policy.expected_red_is_confirmed({**RED, "outer_timeout": True}), False)

    def test_unknown_database_cleanup_cannot_be_expected_red(self):
        self.assertIs(policy.expected_red_is_confirmed({**RED, "database_cleanup_confirmed": False}), False)

    def test_signal_death_cannot_be_expected_red(self):
        self.assertIs(policy.expected_red_is_confirmed({**RED, "returncode": -9}), False)

    def test_missing_registration_cannot_confirm_orphan_cleanup(self):
        self.assertIs(policy.children_gone_is_confirmed([IDENTITY], []), False)

    def test_alive_exact_identity_cannot_confirm_orphan_cleanup(self):
        self.assertIs(policy.children_gone_is_confirmed([IDENTITY], [IDENTITY]), False)

    def test_independently_observed_gone_identity_can_confirm_non_running_only(self):
        self.assertIs(policy.children_gone_is_confirmed([IDENTITY], [{"pid": 12345, "start_ticks": 67890, "state": "gone"}]), True)

    def test_output_exact_cap_is_accepted(self):
        self.assertEqual(policy.bounded_output([32768, 32768], 65536), {"retained_bytes": 65536, "overflow": False, "drain_remaining": True})

    def test_output_overflow_is_failure_with_continued_drain(self):
        self.assertEqual(policy.bounded_output([65536, 1, 999], 65536), {"retained_bytes": 65536, "overflow": True, "drain_remaining": True})
